La fonction $f$ définie sur $\R$ par $f(x)=(3 x-15)(x+2)$ admet pour tableau de signes :

\smallskip

\newcommand\gentblsg[3]{%
	\begin{tikzpicture}[baseline=(current bounding box.center),scale=0.5,transform shape]
		\tkzTabInit[lgt=1.5,espcl=1.5]{$x$/0.8,$f(x)$/0.8}{$-\infty$,$#1$,$#2$,$+\infty$}
		\tkzTabLine{#3}
	\end{tikzpicture}%
}
\eamreponsesautom[2]%
	{\gentblsg{-2}{5}{,+,z,-,z,+,}}%
	{\gentblsg{-2}{5}{,-,z,+,z,-,}}%
	{\gentblsg{-5}{2}{,+,z,-,z,+,}}%
	{\gentblsg{-5}{2}{,-,z,+,z,-,}}%
